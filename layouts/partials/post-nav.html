{{- /* Older/newer post links with cover thumbnails, shown at the end of a post. Selected by date explicitly so the direction never depends on Hugo's Prev/Next sort conventions. */ -}} {{- $posts := where site.RegularPages "Type" "posts" -}} {{- $older := first 1 (where $posts "Date" "lt" .Date).ByDate.Reverse -}} {{- $newer := first 1 (where $posts "Date" "gt" .Date).ByDate -}} {{- if or $older $newer }}
{{- range $older }} {{- with .Param "cover.image" }} {{- end }} ← Previous {{ .Title }} {{- end }} {{- range $newer }} {{- with .Param "cover.image" }} {{- end }} Next → {{ .Title }} {{- end }}
{{- end }}
